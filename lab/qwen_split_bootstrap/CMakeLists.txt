cmake_minimum_required(VERSION 3.24)
project(qwen_copied_route_host_tests LANGUAGES CXX)
if(NOT STRIX_HOST_CORE_LIBRARY)
  message(FATAL_ERROR "Pass the actual host strix_core library path")
endif()
set(root "${CMAKE_CURRENT_LIST_DIR}/../..")
# Set NDEBUG only on the explicitly named target, including Release builds.
string(REPLACE "/DNDEBUG" "" CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE}")
string(REPLACE "-DNDEBUG" "" CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE}")
add_library(actual_core STATIC IMPORTED)
set_target_properties(actual_core PROPERTIES IMPORTED_LOCATION "${STRIX_HOST_CORE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${root}/include")
find_package(Threads REQUIRED)
enable_testing()
foreach(mode IN ITEMS normal ndebug)
  foreach(case IN ITEMS copied_host_channel copied_model_retirement)
    add_executable(${case}-${mode} "${root}/tests/${case}_tests.cpp")
    target_link_libraries(${case}-${mode} PRIVATE actual_core Threads::Threads)
    target_compile_features(${case}-${mode} PRIVATE cxx_std_20)
    if(MSVC)
      target_compile_options(${case}-${mode} PRIVATE /W4 /WX /EHsc /fp:strict)
      if(mode STREQUAL "normal")
        target_compile_options(${case}-${mode} PRIVATE /UNDEBUG)
      else()
        target_compile_definitions(${case}-${mode} PRIVATE NDEBUG)
      endif()
    else()
      target_compile_options(${case}-${mode} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
      if(mode STREQUAL "normal")
        target_compile_options(${case}-${mode} PRIVATE -UNDEBUG)
      else()
        target_compile_definitions(${case}-${mode} PRIVATE NDEBUG)
      endif()
    endif()
    add_test(NAME ${case}-${mode} COMMAND ${case}-${mode})
  endforeach()
endforeach()

# Optional ACTUAL factory ownership test. Explicit declaration-only headers are
# required and this target never enters the native provider build graph.
if(STRIX_NONNATIVE_HIP_DECLARATIONS)
  foreach(mode IN ITEMS normal ndebug)
    set(factory_target factory-owner-${mode}-NONnative)
    add_executable(${factory_target} factory_owner_host_tests.cpp "${root}/src/hip_copied_host_sequence.cpp")
    target_link_libraries(${factory_target} PRIVATE actual_core Threads::Threads)
    target_include_directories(${factory_target} PRIVATE "${STRIX_NONNATIVE_HIP_DECLARATIONS}")
    target_compile_features(${factory_target} PRIVATE cxx_std_20)
    if(mode STREQUAL "ndebug")
      target_compile_definitions(${factory_target} PRIVATE NDEBUG)
    endif()
    if(MSVC)
      target_compile_options(${factory_target} PRIVATE /W4 /WX /EHsc /wd4324)
      target_compile_definitions(${factory_target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_test(NAME ${factory_target} COMMAND ${factory_target})
  endforeach()
endif()
