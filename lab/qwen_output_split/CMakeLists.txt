cmake_minimum_required(VERSION 3.24)
project(qwen_output_split_host LANGUAGES CXX)
enable_testing()
add_executable(strix-qwen-output-split-host host_tests.cpp)
target_compile_features(strix-qwen-output-split-host PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(strix-qwen-output-split-host PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(strix-qwen-output-split-host PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror -ffp-contract=off)
endif()
add_test(NAME qwen-output-split-host COMMAND strix-qwen-output-split-host)
