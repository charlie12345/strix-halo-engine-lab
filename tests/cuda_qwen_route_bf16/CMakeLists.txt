cmake_minimum_required(VERSION 3.24)
project(qwen_route_bf16_host LANGUAGES CXX)
enable_testing()
foreach(name IN ITEMS checks ndebug)
    add_executable(qwen-route-bf16-${name} host.cpp)
    target_include_directories(qwen-route-bf16-${name} PRIVATE ../../include)
    target_compile_features(qwen-route-bf16-${name} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_definitions(qwen-route-bf16-${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(qwen-route-bf16-${name} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
    else()
        target_compile_options(qwen-route-bf16-${name} PRIVATE -Wall -Wextra -Werror -Wconversion -Wsign-conversion -pedantic)
    endif()
    if(name STREQUAL "ndebug")
        target_compile_definitions(qwen-route-bf16-${name} PRIVATE NDEBUG=1)
    endif()
    add_test(NAME qwen-route-bf16-${name} COMMAND qwen-route-bf16-${name})
endforeach()
add_executable(qwen-route-bf16-annotation annotation.cpp)
target_include_directories(qwen-route-bf16-annotation PRIVATE ../../include)
target_compile_features(qwen-route-bf16-annotation PRIVATE cxx_std_20)
add_test(NAME qwen-route-bf16-annotation COMMAND qwen-route-bf16-annotation)

# Compile the actual execution diagnostic as C++ to validate its host API and
# file writer. This object is not linked/run and does not compile device code.
add_library(qwen-route-bf16-e2e-host-syntax OBJECT ../cuda/qwen_route_bf16_e2e.cu)
set_source_files_properties(../cuda/qwen_route_bf16_e2e.cu PROPERTIES LANGUAGE CXX)
target_include_directories(qwen-route-bf16-e2e-host-syntax PRIVATE ../../include)
target_compile_features(qwen-route-bf16-e2e-host-syntax PRIVATE cxx_std_20)
if(MSVC)
    target_compile_definitions(qwen-route-bf16-e2e-host-syntax PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(qwen-route-bf16-e2e-host-syntax PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endif()

# Existing real prefix identity/store regressions, without a CUDA runtime.
add_executable(qwen-route-bf16-prefix-host
    ../cuda/qwen_prefix_cache_host_test.cpp
    ../../src/cuda/qwen_prefix_identity.cpp ../../src/prefix_state_cache.cpp
    ../../src/model_content_hash.cpp ../../src/model_runtime_binding.cpp)
target_compile_features(qwen-route-bf16-prefix-host PRIVATE cxx_std_20)
target_include_directories(qwen-route-bf16-prefix-host PRIVATE ../../include)
target_compile_definitions(qwen-route-bf16-prefix-host PRIVATE
    STRIX_PREFIX_TEST_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/../.." NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(qwen-route-bf16-prefix-host PRIVATE /EHsc /utf-8 /Zc:preprocessor)
endif()
add_test(NAME qwen-route-bf16-prefix-host COMMAND qwen-route-bf16-prefix-host)
